def analyze_person(name, packs):
    print(f"--- Analysis for {name} ---")
    all_colors = ["Green", "Blue", "Orange", "Red", "Yellow"]
    
    # Pack by pack analysis
    pack_summaries = []
    overall_counts = {color: 0 for color in all_colors}
    total_all_sweets = 0
    
    for i, pack in enumerate(packs):
        total_sweets = sum(pack.values())
        total_all_sweets += total_sweets
        print(f"Pack {i+1} (Total: {total_sweets})")
        print("| Color | f | P(Color) | % |")
        print("| :--- | :--- | :--- | :--- |")
        for color in all_colors:
            f = pack[color]
            prob = f / total_sweets
            perc = prob * 100
            overall_counts[color] += f
            print(f"| {color} | {f} | {prob:.3f} | {perc:.1f}% |")
        print()
    
    # Overall analysis
    print(f"Overall Distribution (Total: {total_all_sweets})")
    print("| Color | Total f | P(Color) | % |")
    print("| :--- | :--- | :--- | :--- |")
    for color in all_colors:
        f = overall_counts[color]
        prob = f / total_all_sweets
        perc = prob * 100
        print(f"| {color} | {f} | {prob:.3f} | {perc:.1f}% |")
    print("\n")

analyze_person("Keyshaun Sookdar", keyshaun_data)
